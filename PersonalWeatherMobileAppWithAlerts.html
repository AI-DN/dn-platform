            async checkExistingPermission() {
                if ('geolocation' in navigator && 'permissions' in navigator) {
                    try {
                        const permission = await navigator.permissions.query({ name: 'geolocation' });
                        this.permissionStatus = permission.state;
                        
                        if (permission.state === 'granted') {
                            this.requestLocation();
                        }
                    } catch (error) {
                        console.log('Permission API not supported, using standard geolocation');
                    }
                }
            }

            showState(stateId) {
                const states = ['permission-request', 'location-loading', 'location-success', 'location-error', 'manual-input'];

            async requestLocation() {
                if (!('geolocation' in navigator)) {
                    this.showError(
                        'Geolocation is not supported by this browser.',
                        'Please enter your location manually or use a different browser.'
                    );
                    return;
                }

                this.showLoading();

                const options = {
                    enableHighAccuracy: true,
                    timeout: 15000,
                    maximumAge: 300000 // 5 minutes
                };

                navigator.geolocation.getCurrentPosition(
                    (position) => this.onLocationSuccess(position),
                    (error) => this.onLocationError(error),
                    options
                );
            }

            async onLocationSuccess(position) {
                const { latitude, longitude } = position.coords;
                
                try {
                    // Simulate reverse geocoding (in real app, use actual geocoding service)
                    const location = await this.reverseGeocode(latitude, longitude);
                    this.showSuccess(location);
                } catch (error) {
                    // If geocoding fails, still show coordinates
                    const location = {
                        latitude,
                        longitude,
                        city: 'Unknown City',
                        state: 'Unknown State'
                    };
                    this.showSuccess(location);
                }
            }

            onLocationError(error) {
                let errorMessage = 'Unable to retrieve your location.';
                let details = '';

                switch (error.code) {
                    case error.PERMISSION_DENIED:
                        errorMessage = 'Location access was denied.';
                        details = 'Please enable location services in your browser settings and try again.';
                        break;
                    case error.POSITION_UNAVAILABLE:
                        errorMessage = 'Location information is unavailable.';
                        details = 'Your device may not support location services or GPS may be disabled.';
                        break;
                    case error.TIMEOUT:
                        errorMessage = 'Location request timed out.';
                        details = 'Please check your internet connection and try again.';
                        break;
                    default:
                        details = 'An unknown error occurred while retrieving your location.';
                        break;
                }

                this.showError(errorMessage, details);
            }

                    handleAddLocation(result.key, result.name);
                }
            };

            if (isLoading) {
                return (
                    <div className="container">
                        <div className="glass loading">
                            <div className="loading-spinner"></div>
                            <p>Loading weather data...</p>
                        </div>
                    </div>
                );
            }

            return (
                <div className="container">
                    <div className="glass header">
                        <h1>Weather Locations</h1>
                        <p>Manage and switch between your saved locations</p>
                    </div>

                    <div className="glass current-location">
                        <h2>
                            <span className="location-icon">📍</span>
                            Current Location
                        </h2>
                        <div className="current-weather">
                            <div className="temp-section">
                                <div className="temperature">{currentWeather.temperature}°</div>
                                <div className="condition">{currentWeather.condition}</div>
                            </div>
                            <div className="weather-icon">{currentWeather.icon}</div>
                        </div>
                        <div className="weather-details">
                            <div className="detail-item">
                                <div className="detail-label">Precipitation</div>
                                <div className="detail-value">{currentWeather.precipitation}%</div>
                            </div>
                            <div className="detail-item">
                                <div className="detail-label">Wind</div>
                                <div className="detail-value">{currentWeather.wind} mph</div>
                            </div>
                            <div className="detail-item">
                                <div className="detail-label">Humidity</div>
                                <div className="detail-value">{currentWeather.humidity}%</div>
                            </div>
                            <div className="detail-item">
                                <div className="detail-label">Pressure</div>
                                <div className="detail-value">{currentWeather.pressure} in</div>
                            </div>
                        </div>
                    </div>

                    <div className="glass saved-locations">
                        <div className="section-header">
                            <h3 className="section-title">Saved Locations</h3>
                            <button 
                                className="add-location-btn"
                                onClick={() => setShowAddForm(!showAddForm)}
                                title="Add Location"
                            >
                                {showAddForm ? '×' : '+'}
                            </button>
                        </div>

                        {showAddForm && (
                            <div className="add-location-form glass">
                                <form onSubmit={handleSearchSubmit}>
                                    <div className="form-group">
                                        <label className="form-label">Search for a city</label>
                                        <input
                                            type="text"
                                            className="form-input"
                                            placeholder="Enter city name..."
                                            value={searchQuery}
                                            onChange={(e) => setSearchQuery(e.target.value)}
                                            autoFocus
                                        />
                                    </div>
                                    
                                    {searchResults.length > 0 && (
                                        <div style={{ marginBottom: '15px' }}>
                                            <div className="form-label">Search Results:</div>
                                            {searchResults.map(result => (
                                                <div 
                                                    key={result.key}
                                                    className="location-item"
                                                    style={{ marginBottom: '8px', cursor: 'pointer' }}
                                                    onClick={() => handleAddLocation(result.key, result.name)}
                                                >
                                                    <div className="location-name">{result.name}</div>
                                                </div>
                                            ))}
                                        </div>
                                    )}
                                    
                                    <div className="form-buttons">
                                        <button 
                                            type="button" 
                                            className="btn btn-secondary"
                                            onClick={() => {
                                                setShowAddForm(false);
                                                setSearchQuery('');
                                            }}
                                        >
                                            Cancel
                                        </button>
                                        <button 
                                            type="submit" 
                                            className="btn btn-primary"
                                            disabled={searchResults.length === 0}
                                        >
                                            Add Location
                                        </button>
                                    </div>
                                </form>
                            </div>
                        )}

                        {savedLocationWeather.length === 0 ? (
                            <div className="empty-state">
                                <div className="empty-state-icon">🌍</div>
                                <p>No saved locations yet</p>
                                <p style={{ fontSize: '0.9rem', opacity: '0.7', marginTop: '8px' }}>
                                    Tap the + button to add your first location
                                </p>
                            </div>
                        ) : (
                            savedLocationWeather.map(location => (
                                <div 
                                    key={location.key}
                                    className={`location-item ${currentLocation === location.key ? 'active' : ''}`}
                                    onClick={() => handleLocationSwitch(location.key)}
                                >
                                    <button 
                                        className="delete-btn"
                                        onClick={(e) => handleRemoveLocation(location.key, e)}
                                        title="Remove location"
                                    >
                                        ×
                                    </button>
                                    <div className="location-header">
                                        <div className="location-name">{location.name}</div>
                                        <div className="location-temp">{location.temperature}°</div>
                                    </div>
                                    <div className="location-details">
                                        <div className="location-condition">
                                            <span>{location.icon}</span>
                                            <span>{location.condition}</span>
                                        </div>
                                        <div>💧 {location.precipitation}% • 💨 {location.wind}mph</div>
                                    </div>
                                </div>
                            ))
                        )}
                    </div>
                </div>
            );
        }

        ReactDOM.render(<MultipleLocationSupport />, document.getElementById('root'));
